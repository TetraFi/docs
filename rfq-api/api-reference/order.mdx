description: "This endpoint provides a simple order placement mechanism for quotes retrieved from [/quote](#/v3/v3_quote_v3_quote_post) endpoint. You will be required to sign the quote object using an EOA private key and POST this signature along with the quote ID to place an order."
openapi: "POST /v3/order"
mode: "wide"
---

TetraFi submits order on chain having received maker and taker signatures. This means that TetraFi pays the network (gas) fees as they are already included in the price.



## OpenAPI

````yaml /specs/rfq-api.json post /v3/order
openapi: 3.1.0
info:
  title: TetraFi RFQ API
  version: '3'
  description: >-
    Request-for-quote trading API. Get quotes and execute token swaps directly
    with market makers.
servers:
  - url: https://api.tetrafi.io/pmm/{chain}
    variables:

        default: ethereum
        enum:

          - polygon
          - arbitrum
          - optimism
          - base
          - bsc
          - blast
          - zksync
          - mode

          - taiko
          - superseed
          - berachain
          - hyperevm

        description: Blockchain network
security:
  - bearerAuth: []
paths:
  /v3/order:
    post:
      summary: Order
      description: >-
        This endpoint provides a simple order placement mechanism for quotes
        retrieved from [/quote](#/v3/v3_quote_v3_quote_post) endpoint. You will
        be required to sign the quote object using an EOA private key and POST
        this signature along with the quote ID to place an order.

                    TetraFi submits order on chain having received maker and taker signatures. This means that TetraFi pays the network (gas) fees as they are already included in the price.
      operationId: submitOrder

        content:
          application/json:

              $ref: '#/components/schemas/Body_post_order_v3_order_post'

            application/json:
              schema:
                $ref: '#/components/schemas/OrderApiResponse'
        '422':
          description: Validation Error

            application/json:
              schema:
                $ref: '#/components/schemas/HTTPValidationError'

    Body_post_order_v3_order_post:

        signature:
          type: string

          description: Sign scheme used to produce the signature

        permit2:
          anyOf:
            - $ref: '#/components/schemas/Permit2'

          title: Permit2 info
          description: Permit2 info for taker
        permit:

            - $ref: '#/components/schemas/Permit'
            - type: 'null'
          title: Permit info

      type: object
      required:
        - quote_id
        - signature
      title: Body_post_order_v3_order_post
    OrderApiResponse:

          title: Txhash
        status:
          type: string
          title: Status

          type: integer
          title: Expiry
      type: object

        - status

      title: OrderApiResponse
      examples:

    HTTPValidationError:
      properties:
        detail:
          items:
            $ref: '#/components/schemas/ValidationError'

          title: Detail
      type: object
      title: HTTPValidationError
    Permit2:

        signature:

          type: integer
          title: Approvals Deadline
        token_addresses:

        token_nonces:
          items:
            type: integer
          type: array
          title: Token Nonces
      type: object
      required:
        - signature
        - approvals_deadline
        - token_addresses
        - token_nonces
      title: Permit2
    Permit:
      properties:
        signature:
          type: string
          title: Signature
        approvals_deadline:

          title: Approvals Deadline
      type: object
      required:
        - signature
        - approvals_deadline
      title: Permit
    ValidationError:
      properties:

          items:

              - type: string

          type: array
          title: Location

          title: Message
        type:

          title: Error Type
      type: object
      required:
        - loc

  securitySchemes:
````
