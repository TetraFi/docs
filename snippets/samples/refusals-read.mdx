/** Run a call that should be refused, and read the refusal as a value. */
async function refusal(call: () => Promise<unknown>) {
  try {
    await call();
    return { refused: false as const };
  } catch (error) {
    if (!isTetrafiApiError(error)) throw error; // No HTTP answer at all is a different problem: rethrow it.
    return { refused: true as const, status: error.status, code: error.code, message: error.message,
      next: error.next, requestId: error.requestId };
  }
}
