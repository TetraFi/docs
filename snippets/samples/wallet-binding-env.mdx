// One wallet per chain: TETRAFI_WALLET_EIP155_84532=<organisation wallet id> becomes
// { 'eip155:84532': '<organisation wallet id>' }.
const wallets: Record<string, string> = {};
for (const [name, value] of Object.entries(process.env)) {
  const match = /^TETRAFI_WALLET_([A-Z0-9]+)_(.+)$/.exec(name);
  if (match?.[1] && match[2] && value) wallets[`${match[1].toLowerCase()}:${match[2]}`] = value;
}
const chains = Object.keys(wallets);
