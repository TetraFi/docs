// Chains are CAIP-2 ids (`eip155:<chain id>` for EVM chains). Read them from the asset list: take two EVM
// chains where USDC is native, and make sure ETH exists on the first for the single-chain swap.
const { assets } = await client.assets();
const chainsOf = (symbol: string): string[] => assets.filter((asset) => asset.symbol === symbol)
  .flatMap((asset) => asset.variants).filter((variant) => variant.kind === 'native')
  .flatMap((variant) => variant.deployments.map((deployment) => deployment.chain));
const ethChains = new Set(chainsOf('ETH'));
const usdcChains = [...new Set(chainsOf('USDC'))].filter((chain) => chain.startsWith('eip155:'));
const from = usdcChains.find((chain) => ethChains.has(chain));
const to = usdcChains.find((chain) => chain !== from);
if (!from || !to) throw new Error('This workspace cannot reach USDC on two EVM chains yet.');
