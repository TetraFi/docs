// Every answer's `sourceOptions` lists, per network of the request, the trading wallets this key may name,
// each by its financialAccountId; `isDefault` marks the one used when you name none. Asked without a chain,
// it lists networks the key has a wallet on.
const walletsIn = (set: QuoteSet): Record<string, string> => {
  const found: Record<string, string> = {};
  for (const option of set.sourceOptions ?? []) {
    if (option.kind !== 'wallet' || !usdcChains.has(option.chain)) continue;
    const chosen = option.candidates.find((candidate) => candidate.isDefault) ?? option.candidates[0];
    if (chosen) found[option.chain] = chosen.financialAccountId;
  }
  return found;
};
let tradingWallets = walletsIn(await client.quote({ sell: { asset: 'USDC' }, buy: { asset: 'ETH' }, sellAmount: '10' }));
// To move USDC between chains, ask each corridor from a wallet's chain until one has a wallet on both sides.
const home = Object.keys(tradingWallets).sort()[0];
for (const other of [...usdcChains].sort()) {
  if (!home || other === home || Object.keys(tradingWallets).length >= 2) break;
  await pause();
  const both = walletsIn(await client.quote({ sell: { asset: 'USDC', chain: home }, buy: { asset: 'USDC', chain: other }, sellAmount: '10' }));
  if (both[home] && both[other]) tradingWallets = { [home]: both[home], [other]: both[other] };
}
const chains = Object.keys(tradingWallets).sort((a, b) => (a === home ? -1 : b === home ? 1 : a.localeCompare(b)));
