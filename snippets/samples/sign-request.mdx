import { createHash, createHmac, randomBytes } from 'node:crypto';

const WRITES = ['POST', 'PUT', 'PATCH', 'DELETE'];

// Returns the headers to add to one request.
// `path` includes the query string exactly as sent; `body` is the exact
// bytes you send ('' for none). A write gets a fresh X-Nonce; a read
// signs without one.
function signRequest(signingSecret, method, path, body = '',
  timestamp = Math.floor(Date.now() / 1000),
  nonce = randomBytes(18).toString('base64url')) {
  const write = WRITES.includes(method);
  const digest = createHash('sha256').update(body).digest('hex');
  const lines = write
    ? [timestamp, nonce, method, path, digest]
    : [timestamp, method, path, digest];
  // The secret is 64 hexadecimal characters: the key is its 32 bytes.
  const key = Buffer.from(signingSecret, 'hex');
  const signature = createHmac('sha256', key)
    .update(lines.join('\n'))
    .digest('hex');
  const headers = {
    'X-Timestamp': String(timestamp),
    'X-Signature': signature,
  };
  if (write) headers['X-Nonce'] = nonce;
  return headers;
}

import hashlib
import hmac
import secrets
import time

WRITES = ("POST", "PUT", "PATCH", "DELETE")


# Returns the headers to add to one request.
# path includes the query string exactly as sent; body is the exact bytes
# you send (b"" for none). A write gets a fresh X-Nonce; a read signs
# without one.
def sign_request(signing_secret, method, path, body=b"",
                 timestamp=None, nonce=None):
    timestamp = int(time.time()) if timestamp is None else timestamp
    write = method in WRITES
    digest = hashlib.sha256(body).hexdigest()
    lines = [str(timestamp)]
    if write:
        nonce = nonce or secrets.token_urlsafe(18)
        lines.append(nonce)
    lines += [method, path, digest]
    # The secret is 64 hexadecimal characters: the key is its 32 bytes.
    key = bytes.fromhex(signing_secret)
    message = "\n".join(lines).encode()
    signature = hmac.new(key, message, hashlib.sha256).hexdigest()
    headers = {"X-Timestamp": str(timestamp), "X-Signature": signature}
    if write:
        headers["X-Nonce"] = nonce
    return headers
