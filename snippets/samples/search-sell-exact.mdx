# Sell exactly `amount` of the sell asset. Each quote states what you would buy.
request=$(jq -nc --arg sell "$sell" --arg buy "$buy" --arg amount "$amount" '{
  sell: {asset: $sell},
  buy: {asset: $buy},
  sellAmount: $amount
}')
sell_exact=$(quote_request "$request")
jq -c '.quotes[] | {venue: .legs[0].venue.id, location: .location.kind,
  commitment: .commitment.kind, sell: .sell.amount, buy: .buy.amount, expiresAt}' <<<"$sell_exact" >&2

// Sell exactly 10 USDC. Each quote says how much ETH you would receive for it.
const sellExact = await client.quote({
  sell: { asset: usdc.assetId },

// An order book trades in whole lot sizes, so it may spend a little less than you asked. It never spends
// more, and `unspent` says how much stays in your account.
