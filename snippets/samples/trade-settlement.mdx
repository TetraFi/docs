```bash cURL
# One venue trade's settlement: the obligations that cover it, each with its venue account, window and
# state. A venue trade settled only through its window has no separate settlement record: GET
# …/trades/{tradeId}/settlement answers a named 404 for it.
if [[ -n ${TETRAFI_TRADE_ID:-} ]]; then
  tetrafi GET "/api/v1/workspaces/$workspace/trades/$TETRAFI_TRADE_ID/settlement/obligations" \
    | jq -c --arg trade "$TETRAFI_TRADE_ID" '{tradeId: $trade, obligations: [.items[] | {windowId, obligationId, state}]}' >&2
fi
```

```typescript TypeScript SDK
// One venue trade's settlement: the obligations that cover it (venue account, window and state), then the
// trade's own settlement record. A venue trade settled only through its window has no separate record and
// answers a named 404, shown here as that refusal.
const tradeId = process.env.TETRAFI_TRADE_ID?.trim() || null;
const trade = tradeId ? {
  tradeId,
  obligations: (await client.listTradeSettlementObligations(tradeId)).items
    .map((item) => ({ windowId: item.windowId, obligationId: item.obligationId, state: item.state })),
  settlement: await readOrRefusal(async () => {
    const settlement = await client.getTradeSettlement(tradeId);
    return settlement.lifecycle.map((stage) => ({ stage: stage.stage, status: stage.status }));
  }),
} : null;
```
