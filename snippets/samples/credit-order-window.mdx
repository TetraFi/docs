/**
 * Orders a card with the settlement window it carries. If the window assigned at the send is no longer that
 * one, the order is refused 409 SETTLEMENT_WINDOW_CHANGED and nothing is sent: show the new window
 * (`details.currentWindow`) to whoever decides, and send the same quote again only if they approve it.
 */
type SettlementWindow = { windowId?: string } & Record<string, unknown>;

async function orderConsentingToWindow(
  client: TetrafiClient,
  quote: Quote,
  windowId: string | null,
  approveWindow: (window: SettlementWindow) => Promise<boolean>,
) {
  try {
    return await client.order({ quoteId: quote.id, ...(windowId ? { settlementWindowId: windowId } : {}) });
  } catch (error) {
    if (!isTetrafiApiError(error) || error.code !== 'SETTLEMENT_WINDOW_CHANGED') throw error;
    const details = error.details as { currentWindow?: SettlementWindow | null } | null;
    const current = details?.currentWindow;
    if (!current?.windowId || !(await approveWindow(current))) throw error;
    return client.order({ quoteId: quote.id, settlementWindowId: current.windowId });
  }
}
