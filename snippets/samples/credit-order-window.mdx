/**
 * Orders a card with the settlement window it carries. If the window assigned at the send is no longer that
 * one, the order is refused 409 SETTLEMENT_WINDOW_CHANGED and nothing is sent: show the new window
 * (`details.currentWindow`) to whoever decides, then send the same quote again consenting to it.
 */
async function orderConsentingToWindow(client: TetrafiClient, quote: Quote, windowId: string | null) {
  try {
    return await client.order({ quoteId: quote.id, ...(windowId ? { settlementWindowId: windowId } : {}) });
  } catch (error) {
    if (!isTetrafiApiError(error) || error.code !== 'SETTLEMENT_WINDOW_CHANGED') throw error;
    const details = error.details as { currentWindow?: { windowId?: string } | null } | null;
    const current = details?.currentWindow?.windowId;
    if (!current) throw error;
    return client.order({ quoteId: quote.id, settlementWindowId: current });
  }
}
