function testSiteCard(set: QuoteSet): Quote | undefined {
  // An indicative card executes only within your own minimum receive, so it is chosen only when you set one.
  return set.quotes.find((card) => card.next.kind === 'order'
    && (card.quality === 'executable' || card.quality === 'firm' || (card.quality === 'indicative' && !!minReceive))
    && (card.expiresAt == null || Date.parse(card.expiresAt) > Date.now())
    && card.legs.every((leg) => leg.execution != null && TEST_SITES.includes(leg.execution.site)));
}
