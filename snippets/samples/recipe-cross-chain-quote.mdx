# The same asset on both sides, each side pinned to its chain, and only cross-chain quotes kept.
request=$(jq -nc --arg asset "$asset" --arg amount "$amount" --arg from "$from_chain" --arg to "$to_chain" '{
  kind: "pair",
  sell: {asset: $asset, network: $from},
  buy: {asset: $asset, network: $to},
  sellAmount: $amount,
  filters: {locations: ["crossChain"]}
}')
quotes=$(quote "$request")

# Every venue asked, with its outcome. apiKeyOrganisationWalletScopeRequired or onchainWalletNotUsable
# means the key has no organisation wallet granted to it that the venue can use.
jq -c '.sources[] | select(.status != "filteredOut") | {venue: .venue.id, status, reason}' <<<"$quotes" >&2

# next "order" with signing.mode "tetrafi": TetraFi executes the move from an organisation wallet, so
# the key can order it. signing.mode "you" is a wallet you sign for (recipe-own-wallet.sh).
# next "preflight": settles through TetraFi's router and escrow (Advanced), not open yet.
jq -c '.quotes[] | {quoteId: .id, venue: .legs[0].venue.id, next: .next.kind,
  sell: "\(.sell.amount) on \(.sell.network)", buy: "\(.buy.amount) on \(.buy.network)",
  eta, tradesFrom: (.tradingSource | {kind, financialAccountId, network}), fundingModel}' <<<"$quotes" >&2
quote=$(jq -c 'first(.quotes[] | select(.next.kind == "order" and .signing.mode != "you")) // empty' <<<"$quotes")
