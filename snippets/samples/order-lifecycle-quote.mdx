// quote is preferred: it can be ordered as it stands.
const orderable = set.quotes.filter((quote) => quote.next.kind === 'order'
  && quote.legs.every((leg) => leg.execution !== null && TEST_SITES.includes(leg.execution.site)));
const card = orderable.find((quote) => quote.commitment.kind === 'executable') ?? orderable[0];
if (!card) throw new Error('No sandbox or testnet venue offered a quote that can be ordered now.');
// The quote's own floor when it has one, otherwise the amount it shows.
const minReceive = card.minOut ?? card.receive.amount;
if (!minReceive) throw new Error(`Quote ${card.id} has no amount to protect, so it is not ordered.`);
