# List the venue accounts (Treasury read); each accountId is a venueAccountId. A wallet-based venue has no
# accountId and no credit line. GET …/venue-accounts/{venueAccountId}/credit reads the line:
# observation.state reported carries limit, used, available and overrun; notReported and unavailable carry
# no figures, never zeros. positions are the signed positions from the last recorded ledger read.
accounts=$(tetrafi GET "/api/v1/workspaces/$workspace/venue-enablements?limit=50")
for id in $(jq -r '.items[] | .accountId // empty' <<<"$accounts" | head -n 5); do
  credit=$(tetrafi GET "/api/v1/workspaces/$workspace/venue-accounts/$id/credit")
  jq -c '{venueAccountId, line: (.observation | if .state == "reported"
      then {limit, used, available, overrun, currency, observedAt} else {state, code: .error} end),
    facility: .facility.state, positions: (.positions | if .state == "observed"
      then [.value[] | {asset, netPosition}] else {state} end), readAt}' <<<"$credit" >&2
done

/**
 * One account's credit line. `observation` is the provider's own report: `reported` carries limit, used,
 * available and overrun; `notReported` and `unavailable` carry no figures, never zeros. `positions` are the
 * signed positions from the last recorded ledger read, with the time they were observed.
 */
type ObservedPositions = Extract<VenueAccountCredit['positions'], { state: 'observed' }>;

function creditSummary(credit: VenueAccountCredit): Record<string, unknown> {
  const observation = credit.observation;
  const line = 'limit' in observation
    ? { limit: observation.limit, used: observation.used, available: observation.available,
        overrun: observation.overrun, currency: observation.currency ?? null, observedAt: observation.observedAt }
    : { state: observation.state, code: 'error' in observation ? observation.error : null };
  const positions = credit.positions.state === 'observed'
    ? (credit.positions as ObservedPositions).value.map((position) => ({ asset: position.asset, netPosition: position.netPosition }))
    : { state: credit.positions.state };
  return { venueAccountId: credit.venueAccountId, line, facility: credit.facility.state, positions, readAt: credit.readAt };
}

/** Every venue account's credit. A wallet-based enablement has no `accountId` and no credit line. */
async function monitorCredit(client: TetrafiClient): Promise<Array<Record<string, unknown>>> {
  const lines: Array<Record<string, unknown>> = [];
  for await (const page of client.listVenueAccountsAll({ limit: 50 })) {
    for (const account of page.items) {
      if (!account.accountId) continue;
      const credit = await readOrRefusal(() => client.getVenueAccountCredit(account.accountId!));
      lines.push(isRefused(credit) ? { venueAccountId: account.accountId, ...credit } : creditSummary(credit));
    }
  }
  return lines;
}
