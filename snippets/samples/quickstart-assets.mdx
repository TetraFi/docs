items='[]' cursor=""
while :; do # follow nextCursor to the last page
  page=$(tetrafi GET "/api/v1/assets?limit=200${cursor:+&cursor=$cursor}")
  items=$(jq -c --argjson seen "$items" '$seen + .items' <<<"$page")
  cursor=$(jq -r '.nextCursor // empty | @uri' <<<"$page")
  [[ -n $cursor ]] || break
done
assets=$(jq -c '{items: .}' <<<"$items")
for asset in "$sell" "$buy"; do
  if ! jq -e --arg a "$asset" 'any(.items[]; .asset == $a)' <<<"$assets" >/dev/null; then
    echo "{\"error\":\"ASSET_NOT_LISTED\",\"asset\":\"$asset\"}" >&2
    exit 1
  fi
done
echo "assets: $(jq '.items | length' <<<"$assets") listed, quoting $sell for $buy" >&2

const assets = { items: [] };
let cursor = '';
do { // follow nextCursor to the last page
  const next = cursor ? `&cursor=${encodeURIComponent(cursor)}` : '';
  const page = await tetrafi('GET', `/api/v1/assets?limit=200${next}`);
  assets.items.push(...page.items);
  cursor = page.nextCursor ?? '';
} while (cursor);
for (const asset of [sell, buy]) {
  if (!assets.items.some((item) => item.asset === asset)) {
    throw new Error(json({ error: 'ASSET_NOT_LISTED', asset }));
  }
}
console.error(`assets: ${assets.items.length} listed,`,
  `quoting ${sell} for ${buy}`);
