```bash cURL
# The account's windows, newest first; filter with state and follow nextCursor until it is null. A ready
# window's statement says its provenance, whether each price was available, and whether the minimum
# settlement makes it due.
windows=$(tetrafi GET "$base/settlement/windows?limit=20")
jq -c '.items[:3][] | {windowId, state, snapshotAt, paymentDeadlineAt, dueState}' <<<"$windows" >&2
window_id=$(jq -r '.items[0].windowId // empty' <<<"$windows")
if [[ -n $window_id ]]; then
  window=$(tetrafi GET "$base/settlement/windows/$window_id")
  jq -c '{windowId, state, provenance: .statement.provenance, prices: [.statement.prices[]?.status],
    minimum: .statement.minimumSettlement.outcome}' <<<"$window" >&2
fi
```

```typescript TypeScript SDK
// The account's windows, newest first, every page. A ready window carries its statement: `provenance` says
// whether it is TetraFi's indicative statement, each price says whether it was available, and
// `minimumSettlement.outcome` says whether the minimum makes the window due.
const windows: Array<Record<string, unknown>> = [];
for await (const page of client.listSettlementWindowsAll(accountId, { limit: 20 })) {
  for (const window of page.items) {
    windows.push({ windowId: window.windowId, state: window.state, snapshotAt: window.snapshotAt,
      paymentDeadlineAt: window.paymentDeadlineAt ?? null, dueState: window.dueState ?? null });
  }
}
const firstWindowId = windows[0]?.windowId as string | undefined;
const detail = firstWindowId ? await client.getSettlementWindow(accountId, firstWindowId) : null;
const statement = detail?.statement ? {
  provenance: detail.statement.provenance,
  prices: detail.statement.prices.map((price) => price.status),
  minimum: detail.statement.minimumSettlement?.outcome ?? null,
} : null;
```
