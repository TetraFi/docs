/**
 * A card that cannot be ordered says why in `sourceRefusal`. On a credit account, `creditRefusal.fix` is the way
 * forward: `reduceAmount` (an order up to `maxAmount` of `asset` fits), `reduceExposure` (already over the limit
 * by `overBy`: only a trade that reduces what the account owes is admitted) or `retryRead` (ask again).
 */
function blockedBy(quote: Quote): Record<string, unknown> | null {
  const refusal = 'sourceRefusal' in quote ? quote.sourceRefusal ?? null : null;
  if (!refusal) return null;
  return { code: refusal.code, fix: refusal.creditRefusal?.fix ?? refusal.fix };
}
