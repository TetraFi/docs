/**
 * The refusals an order on a credit or prefunded account can meet, read as values. Nothing was sent on any of
 * them; each names what passes.
 */
function orderRefusal(error: TetrafiApiError): Record<string, unknown> {
  const headroom = prefundedHeadroomOf(error); // 409 VENUE_PREFUNDED_BALANCE_INSUFFICIENT
  if (headroom) return { code: error.code, retryWithDebitAtMost: headroom.maxAmount, asset: headroom.asset };
  const credit = creditRefusalOf(error); // 409 CREDIT_LIMIT_BREACH, 503 CREDIT_READ_UNAVAILABLE
  if (credit) return { code: error.code, fix: credit.fix ?? null, shortfall: credit.shortfall ?? null };
  if (error.code === 'LP_CREDIT_EXHAUSTED') return { code: error.code, fix: 'lower the amount, or choose another quote' };
  return { code: error.code, status: error.status, message: error.message };
}
