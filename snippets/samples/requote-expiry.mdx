// How long a quote can still be refreshed after it lapses.
const REFRESH_WINDOW_MS = 10 * 60_000;

/** What to do with a held quote right now. */
function nextStep(quote: Quote, now = Date.now()): 'order' | 'requote' | 'quoteAgain' {
  // An indicative quote can carry no expiry: it is a price to compare, and ordering it always asks again.
  if (quote.expiresAt === null) return 'order';
  const expiresAt = Date.parse(quote.expiresAt);
  if (now < expiresAt) return 'order';
  if (now < expiresAt + REFRESH_WINDOW_MS) return 'requote';
  return 'quoteAgain';
}
