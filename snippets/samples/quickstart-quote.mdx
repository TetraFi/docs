```bash cURL
request=$(jq -n --arg sell "$sell" --arg buy "$buy" --arg amount "$amount" \
  --arg venues "${TETRAFI_VENUES:-}" '{
    sell: {asset: $sell},
    buy: {asset: $buy},
    sellAmount: $amount
  } + (if $venues == "" then {} else {filters: {venues: {include: ($venues | split(","))}}} end)')
quotes=$(tetrafi POST /api/v1/quotes "$request")

# Quotes arrive best first (most received, net of fees). Take the best one you can act on now.
quote=$(jq -c 'first(.quotes[] | select(.next.kind == "order" or .next.kind == "preflight")) // empty' <<<"$quotes")
if [[ -z $quote ]]; then
  # Every venue asked is listed in sources with its outcome, so an empty result says why.
  jq -c '{error: "NO_ACTIONABLE_QUOTE", sources: [.sources[] | {venue: .venue.id, status, reason}]}' <<<"$quotes" >&2
  exit 1
fi
quote_id=$(jq -r '.id' <<<"$quote")
jq -c '{quoteId: .id, venue: .legs[0].venue.id, mode: .legs[0].venue.integrationMode,
  sell: .sell.amount, buy: .buy.amount, next: .next.kind, expiresAt}' <<<"$quote" >&2
```

```typescript TypeScript SDK
const set = await client.quote({
  sell: { asset: usdc.assetId },
  buy: { asset: eth.assetId },
  sellAmount: '10',
});
// Every venue that was considered says what happened, including the ones that gave no quote and why.
const notQuoted = set.sources.filter((source) => source.status !== 'answered')
  .map((source) => `${source.venue.id}: ${source.status}`);
// Quotes arrive best first. Take the first one you can act on right now.
const best: Quote | undefined = set.quotes.find((quote) => quote.next.kind === 'order' || quote.next.kind === 'preflight');
if (!best) throw new Error(`No quote can be ordered yet. Not quoted: ${notQuoted.join('; ') || 'none'}`);
```
