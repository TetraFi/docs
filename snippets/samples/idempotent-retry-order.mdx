let placed: OrderResult;
try {
  placed = await client.order(request, { idempotencyKey });
} catch (error) {
  // The order was placed, but its answer could not be read: send the same request and key to read it.
  if (error instanceof TetrafiOrderUnreadableError) {
    return { ordered: true, answer: 'unreadable', idempotencyKey, request };
  }
  // The SDK already retried what is safe to retry. If the outcome is still unknown, the order may exist:
  // the same request with the same key, sent again later, tells you which.
  if ((error instanceof TetrafiApiError || error instanceof TetrafiNetworkError) && error.outcomeUnknown) {
    return { ordered: 'unknown', idempotencyKey, request, reason: error.message };
  }
  throw error; // Any other refusal is final: nothing was placed.
}
