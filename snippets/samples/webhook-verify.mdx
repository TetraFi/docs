# verify SECRET ID TIMESTAMP BODY SIGNATURES [NOW]
# SECRET is the whsec_… signing secret. ID, TIMESTAMP and SIGNATURES are the webhook-id,
# webhook-timestamp and webhook-signature headers; BODY is the raw request body, before any
# parsing. Succeeds when one signature matches and the timestamp is within 5 minutes of NOW.
verify() {
  local secret=$1 id=$2 timestamp=$3 body=$4 signatures=$5 now=${6:-$(date +%s)}
  (( timestamp > now - 300 && timestamp < now + 300 )) || return 1
  local key expected candidate
  key=$(printf '%s' "${secret#whsec_}" | openssl base64 -d -A | od -An -tx1 | tr -d ' \n')
  expected=$(printf '%s.%s.%s' "$id" "$timestamp" "$body" \
    | openssl dgst -sha256 -mac HMAC -macopt "hexkey:$key" -binary | openssl base64 -A)
  # During a secret rotation the header holds two space-separated signatures: either may match.
  for candidate in $signatures; do
    [[ $candidate == "v1,$expected" ]] && return 0
  done
  return 1
}

function verifyWebhook(secret, headers, rawBody, now = Math.floor(Date.now() / 1000)) {
  const id = headers['webhook-id'];
  const timestamp = Number(headers['webhook-timestamp']);
  if (!id || !Number.isInteger(timestamp) || Math.abs(now - timestamp) >= 300) return false;
  const key = Buffer.from(secret.replace(/^whsec_/, ''), 'base64');
  const expected = createHmac('sha256', key).update(`${id}.${timestamp}.${rawBody}`).digest();
  // During a secret rotation the header holds two space-separated signatures: either may match.
  return String(headers['webhook-signature'] ?? '').split(' ').some((value) => {
    const [version, signature] = value.split(',');
    const given = Buffer.from(signature ?? '', 'base64');
    return version === 'v1' && given.length === expected.length && timingSafeEqual(given, expected);
  });
}

# Pass the raw request body as bytes, before any JSON parsing, and the three webhook-* headers.
# Returns True when one signature matches and the timestamp is within 5 minutes of now.
def verify_webhook(secret, headers, raw_body, now=None):
    now = int(time.time()) if now is None else now
    msg_id = headers.get("webhook-id")
    try:
        timestamp = int(headers.get("webhook-timestamp", ""))
    except ValueError:
        return False
    if not msg_id or abs(now - timestamp) >= 300:
        return False
    key = base64.b64decode(secret[len("whsec_"):] if secret.startswith("whsec_") else secret)
    signed = f"{msg_id}.{timestamp}.".encode() + raw_body
    expected = base64.b64encode(hmac.new(key, signed, hashlib.sha256).digest()).decode()
    # During a secret rotation the header holds two space-separated signatures: either may match.
    for value in headers.get("webhook-signature", "").split(" "):
        version, _, signature = value.partition(",")
        if version == "v1" and hmac.compare_digest(signature, expected):
            return True
    return False
